cmake_minimum_required(VERSION 3.20)
project(market_data_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(market_data src/feed.cpp)
target_include_directories(market_data PUBLIC include)
target_compile_options(market_data PRIVATE -Wall -Wextra -Wpedantic)
add_executable(feed_demo src/main.cpp)
target_link_libraries(feed_demo PRIVATE market_data)
enable_testing()
add_executable(feed_tests tests/test_feed.cpp)
target_link_libraries(feed_tests PRIVATE market_data)
add_test(NAME feed_tests COMMAND feed_tests)
add_executable(spsc_queue_tests tests/test_spsc_queue.cpp)
target_include_directories(spsc_queue_tests PRIVATE include)
add_test(NAME spsc_queue_tests COMMAND spsc_queue_tests)

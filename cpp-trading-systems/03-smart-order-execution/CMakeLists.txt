cmake_minimum_required(VERSION 3.20)
project(smart_order_execution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(execution src/execution.cpp)
target_include_directories(execution PUBLIC include)
target_compile_options(execution PRIVATE -Wall -Wextra -Wpedantic)
add_executable(execution_demo src/main.cpp)
target_link_libraries(execution_demo PRIVATE execution)
enable_testing()
add_executable(execution_tests tests/test_execution.cpp)
target_link_libraries(execution_tests PRIVATE execution)
add_test(NAME execution_tests COMMAND execution_tests)

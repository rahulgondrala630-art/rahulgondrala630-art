cmake_minimum_required(VERSION 3.20)
project(order_matching_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(matching_engine src/order_book.cpp)
target_include_directories(matching_engine PUBLIC include)
target_compile_options(matching_engine PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(matching_engine PUBLIC Threads::Threads)
add_executable(matching_demo src/main.cpp)
target_link_libraries(matching_demo PRIVATE matching_engine)
enable_testing()
add_executable(matching_tests tests/test_order_book.cpp)
target_link_libraries(matching_tests PRIVATE matching_engine)
add_test(NAME matching_tests COMMAND matching_tests)
